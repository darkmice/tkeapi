One Line of Code to Access a Unified API Relay

Tkeapi is compatible with mainstream OpenAI SDKs. Replace the Base URL and API key to call text, image, video, and multimodal models through one unified endpoint, with either direct API access or API subscription plans. 

import openai

client = openai.OpenAI(
    # Replace with Tkeapi relay service base URL
    base_url="__CURRENT_SITE_ORIGIN__/v1",
    # Replace with your Tkeapi API Key
    api_key="sk-tkeapi-xxxxxx"
)

response = client.chat.completions.create(
    model="seedance-2-0-pro",
    messages=[{"role": "user", "content": "Hello, introduce Tkeapi"}],
    stream=True
)
import { OpenAI } from "openai";

const openai = new OpenAI({
  // Replace with Tkeapi API base URL
  baseURL: "__CURRENT_SITE_ORIGIN__/v1",
  // Replace with your Tkeapi API Key
  apiKey: "sk-tkeapi-xxxxxx",
});

const response = await openai.chat.completions.create({
  model: "seedance-2-0-pro",
  messages: [{ role: "user", content: "Hello" }],
  stream: true,
});
curl __CURRENT_SITE_ORIGIN__/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer sk-tkeapi-xxxxxx" \
  -d '{
    "model": "seedance-2-0-pro",

  }'
